219 [[nodiscard]] int16_t readTemperature(const uint8_t idx) const
220 {
221 static ScratchPad buf;
222
223 if constexpr (TEMP_SENSOR_PRESENT)
224 {
225 if (!oneWire.reset())
226 {
227 return DEVICE_DISCONNECTED_RAW;
228 }
229 oneWire.select(sensorAddrs[idx].addr);
230 oneWire.write(READ_SCRATCHPAD);
231
232 for (auto& buf_elem : buf)
233 {
234 buf_elem = oneWire.read();
235 }
236
237 if (!oneWire.reset())
238 {
239 return DEVICE_DISCONNECTED_RAW;
240 }
241 if (oneWire.crc8(buf, 8) != buf[8])
242 {
243 return DEVICE_DISCONNECTED_RAW;
244 }
245 }
246
247 // result is temperature x16, multiply by 6.25 to convert to temperature x100
248 int16_t result = (buf[1] << 8) | buf[0];
249 result = (result * 6) + (result >> 2);
250 if (result <= TEMP_RANGE_LOW || result >= TEMP_RANGE_HIGH)
251 {
252 return OUTOFRANGE_TEMPERATURE; // return value ('Out of range')
253 }
254
255 return result;
256 }
