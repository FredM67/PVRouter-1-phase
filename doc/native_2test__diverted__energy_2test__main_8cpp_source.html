141void test_average_watts(void)
142{
143 // 250 cycles (5 s at 50 Hz), 16 IEU per W per cycle, at most 600000 IEU per cycle
144 constexpr DivertedPower::Average average{ DivertedPower::averageOver(250, 250UL * 600000UL, 4) };
145
146 TEST_ASSERT_EQUAL_INT16(0, DivertedPower::watts(0, average));
147 TEST_ASSERT_EQUAL_INT16(1000, DivertedPower::watts(250UL * 16000UL, average)); // 1000 W all along
148 TEST_ASSERT_EQUAL_INT16(500, DivertedPower::watts(125UL * 16000UL, average)); // half the cycles forced: 0 W
149 TEST_ASSERT_EQUAL_INT16(32767, DivertedPower::watts(250UL * 600000UL, average)); // 37500 W: saturated
150 TEST_ASSERT_EQUAL_INT16(2345, DivertedPower::watts(250UL * 16UL * 2345UL, average));
151
152 // 40 s at 60 Hz, the longest period
153 constexpr DivertedPower::Average longest{ DivertedPower::averageOver(2400, 2400UL * 600000UL, 4) };
154 TEST_ASSERT_EQUAL_INT16(1234, DivertedPower::watts(2400UL * 16UL * 1234UL, longest));
155 TEST_ASSERT_EQUAL_INT16(7, DivertedPower::watts(2400UL * 16UL * 7UL, longest));
156}
