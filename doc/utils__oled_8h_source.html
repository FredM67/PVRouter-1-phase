140inline void get_tile_from_xbm(uint8_t (&tile)[8], uint8_t tx, uint8_t ty, uint8_t xbm_byte_width, const unsigned char *xbm)
141{
142 const auto *p{ xbm + tx + xbm_byte_width * (ty << 3) };
143
144 memset(tile, 0, sizeof(tile));
145
146 for (uint8_t i = 0; i < 8; ++i)
147 {
148 const uint8_t b = u8x8_pgm_read(p);
149 for (uint8_t bit = 0; bit < 8; ++bit)
150 {
151 if (b & (1 << bit))
152 {
153 tile[bit] |= (1 << i);
154 }
155 }
156 p += xbm_byte_width;
157 }
158}

171inline void u8x8_draw_xbm(uint8_t _tx, uint8_t _ty, uint8_t xbm_width, uint8_t xbm_height, const unsigned char *xbm)
172{
173 uint8_t tile[8];
174 for (uint8_t y = 0; y < (xbm_height >> 3); ++y)
175 {
176 for (uint8_t x = 0; x < (xbm_width >> 3); ++x)
177 {
178 get_tile_from_xbm(tile, x, y, xbm_width >> 3, xbm);
179 u8x8.drawTile(_tx + x, _ty + y, 1, tile);
180 }
181 }

243inline void updateOLED(const Display::Item &item, const uint16_t energyWh, const int16_t temperature_x100)
244{
245 if constexpr (TYPE_OF_DISPLAY == DisplayType::OLED)
246 {
247 static bool logoShown{ true };
248 static Display::Item shown{ Display::Content::WAITING, 0xFF };
249 static char shownText[Display::OLED_TEXT_SIZE]{};
250
251 char text[Display::OLED_TEXT_SIZE]{};
252 switch (item.content)
253 {
254 case Display::Content::ENERGY:
255 Display::oledEnergy(energyWh, text);
256 break;
257 case Display::Content::TEMPERATURE:
258 Display::oledTemperature(temperature_x100, text);
259 break;
260 default:
261 break;
262 }
263
264 const bool newItem{ logoShown || (item.content != shown.content) || (item.sensor != shown.sensor) };
265
266 if (logoShown)
267 {
268 logoShown = false;
269 u8x8.clearDisplay();
270 }
271 else if (newItem)
272 {
273 if (text[0] == '\0')
274 {
275 for (uint8_t row = 0; row != 6; ++row) // no large digits to write over the previous ones
276 {
277 u8x8.clearLine(row);
278 }
279 }
280 u8x8.clearLine(6); // the labels; the watchdog comes back below
281 u8x8.clearLine(7);
282 }
283
284 if (newItem)
285 {
286 u8x8.setFont(u8x8_font_7x14B_1x2_r);
287 switch (item.content)
288 {
289 case Display::Content::ENERGY:
290 u8x8.drawString(12, 6, "kWh");
291 break;
292 case Display::Content::TEMPERATURE:
293 {
294 const char sensor[]{ 'T', static_cast< char >('1' + item.sensor), 0 };
295 u8x8.drawString(3, 6, sensor);
296
297 uint8_t degree[8]{ 0x00, 0x06, 0x09, 0x09, 0x06, 0x00, 0x00, 0x00 }; // a small circle, top left
298 u8x8.drawTile(13, 6, 1, degree);
299 u8x8.drawString(14, 6, "C");
300 break;
301 }
302 case Display::Content::FORCED:
303 u8x8.drawString(5, 2, "FORCED");
304 break;
305 case Display::Content::OFF:
306 u8x8.drawString(6, 2, "OFF");
307 break;
308 default:
309 break;
310 }
311 }
312
313 if (text[0] != '\0')
314 {
315 u8x8.setFont(u8x8_font_inb33_3x6_n);
316 for (uint8_t i = 0; i != Display::OLED_TEXT_SIZE - 1; ++i)
317 {
318 if (newItem || (text[i] != shownText[i]))
319 {
320 u8x8.drawGlyph(3 * i, 0, static_cast< uint8_t >(text[i]));
321 }
322 }
323 }
324
325 shown = item;
326 memcpy(shownText, text, sizeof(text));
327
328 updateWatchdog();
329 }
330}
