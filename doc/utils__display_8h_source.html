86inline void setDisplayCharacters(const uint8_t (&chars)[Display::NO_OF_DIGITS])
87{
88 if constexpr (SEVEN_SEGMENT_DISPLAY)
89 {
90 static constexpr SevenSegment::CharImages table PROGMEM{ SevenSegment::makeCharImages() };
91
92 Display::PortImage images[Display::NO_OF_DIGITS];
93 for (uint8_t i = 0; i != Display::NO_OF_DIGITS; ++i)
94 {
95 const uint8_t c{ static_cast< uint8_t >(chars[i] & 0x7F) };
96 Display::PortImage part;
97 memcpy_P(&images[i], &table.digit[i], sizeof(Display::PortImage));
98 memcpy_P(&part, &table.character[c < Display::NO_OF_CHARS ? c : Display::CHAR_BLANK], sizeof(Display::PortImage));
99 images[i] |= part;
100 if (chars[i] & Display::DP)
101 {
102 memcpy_P(&part, &table.dp, sizeof(Display::PortImage));
103 images[i] |= part;
104 }
105 }
106
107 ATOMIC_BLOCK(ATOMIC_RESTORESTATE)
108 {
109 memcpy(SevenSegment::digitImages, images, sizeof(images));
110 }
111 }
112}

119inline void initializeDisplay()
120{
121 if constexpr (SEVEN_SEGMENT_DISPLAY)
122 {
123 using namespace SevenSegment;
124
125 // every pin LOW but every digit off (SEG) or the 74HC4543 off (SEG_HW), then as outputs
126 // (before the ADC interrupt starts: nothing else writes these ports yet)
127 PORTB = static_cast< uint8_t >((PORTB & static_cast< uint8_t >(~PINS.b)) | BLANK_HIGH.b);
128 PORTC = static_cast< uint8_t >((PORTC & static_cast< uint8_t >(~PINS.c)) | BLANK_HIGH.c);
129 PORTD = static_cast< uint8_t >((PORTD & static_cast< uint8_t >(~PINS.d)) | BLANK_HIGH.d);
130 DDRB |= PINS.b;
131 DDRC |= PINS.c;
132 DDRD |= PINS.d;
133
134 const uint8_t blank[Display::NO_OF_DIGITS]{ Display::CHAR_BLANK, Display::CHAR_BLANK, Display::CHAR_BLANK, Display::CHAR_BLANK };
135 setDisplayCharacters(blank);
136 }
137}

150inline void showOn7SegmentDisplay(const Display::Item& item, const uint16_t energyWh, const int16_t temperature_x100)
151{
152 if constexpr (SEVEN_SEGMENT_DISPLAY)
153 {
154 static uint8_t locationOfDot{ 0 };
155 uint8_t chars[Display::NO_OF_DIGITS]{};
156
157 switch (item.content)
158 {
159 case Display::Content::FORCED:
160 Display::segForced(chars);
161 break;
162 case Display::Content::OFF:
163 Display::segOff(chars);
164 break;
165 case Display::Content::ENERGY:
166 Display::segEnergy(energyWh, chars);
167 break;
168 case Display::Content::TEMPERATURE:
169 Display::segTemperature(temperature_x100, TYPE_OF_DISPLAY == DisplayType::SEG, chars);
170 break;
171 default:
172 Display::segWalkingDots(locationOfDot, chars);
173 locationOfDot = (locationOfDot + 1) & 3;
174 break;
175 }
176 setDisplayCharacters(chars);
177 }

197inline void refresh7SegDisplay()
198{
199 if constexpr (SEVEN_SEGMENT_DISPLAY)
200 {
201 using namespace SevenSegment;
202
203 static uint8_t displayTime_count{ 0 };
204 static uint8_t digitLocationThatIsActive{ 0 };
205
206 if (++displayTime_count != MAX_DISPLAY_TIME_COUNT)
207 {
208 return;
209 }
210 displayTime_count = 0;
211
212 digitLocationThatIsActive = (digitLocationThatIsActive + 1) & (Display::NO_OF_DIGITS - 1);
213 const Display::PortImage& image{ digitImages[digitLocationThatIsActive] };
214
215 // 1. nothing shows while the ports change
216 if constexpr (BLANK_HIGH.b != 0)
217 {
218 PORTB |= BLANK_HIGH.b;
219 }
220 if constexpr (BLANK_HIGH.c != 0)
221 {
222 PORTC |= BLANK_HIGH.c;
223 }
224 if constexpr (BLANK_HIGH.d != 0)
225 {
226 PORTD |= BLANK_HIGH.d;
227 }
228 if constexpr (BLANK_LOW.b != 0)
229 {
230 PORTB &= static_cast< uint8_t >(~BLANK_LOW.b);
231 }
232 if constexpr (BLANK_LOW.c != 0)
233 {
234 PORTC &= static_cast< uint8_t >(~BLANK_LOW.c);
235 }
236 if constexpr (BLANK_LOW.d != 0)
237 {
238 PORTD &= static_cast< uint8_t >(~BLANK_LOW.d);
239 }
240
241 // 2. the next digit (SEG_HW: port D last, it holds the enable line)
242 if constexpr (PINS.b != 0)
243 {
244 PORTB = static_cast< uint8_t >((PORTB & static_cast< uint8_t >(~PINS.b)) | image.b);
245 }
246 if constexpr (PINS.c != 0)
247 {
248 PORTC = static_cast< uint8_t >((PORTC & static_cast< uint8_t >(~PINS.c)) | image.c);
249 }
250 if constexpr (PINS.d != 0)
251 {
252 PORTD = static_cast< uint8_t >((PORTD & static_cast< uint8_t >(~PINS.d)) | image.d);
253 }
254 }
255}
