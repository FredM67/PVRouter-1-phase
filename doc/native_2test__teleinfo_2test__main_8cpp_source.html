115class Reference
116{
117private:
118 static const char STX{ 0x02 };
119 static const char ETX{ 0x03 };
120 static const char LF{ 0x0A };
121 static const char CR{ 0x0D };
122 static const char TAB{ 0x09 };

124 char buffer[512]{};
125 size_t bufferPos{ 0 };
126
127 [[nodiscard]] uint8_t calculateChecksum(size_t startPos, size_t endPos) const
128 {
129 uint8_t sum{ 0 };
130 for (const char* ptr = buffer + startPos; ptr != buffer + endPos; ++ptr)
131 {
132 sum += *ptr;
133 }
134
135 return (sum & 0x3F) + 0x20;
136 }
137
138 void writeTag(const char* tag, uint8_t index)
139 {
140 auto* ptr{ tag };
141 while (*ptr) buffer[bufferPos++] = *ptr++;
142
143 if (index != 0)
144 {
145 buffer[bufferPos++] = static_cast< char >('0' + index);
146 }
147
148 buffer[bufferPos++] = TAB;
149 }
150
151 template< typename Magnitude >
152 void writeLine(const char* tag, uint8_t index, Magnitude magnitude, bool negative)
153 {
154 buffer[bufferPos++] = LF;
155
156 const auto startPos{ bufferPos };
157
158 writeTag(tag, index);
159 if (negative)
160 {
161 buffer[bufferPos++] = '-';
162 }
163 char* str{ buffer + bufferPos };
164 snprintf(str, 12, "%lu", static_cast< unsigned long >(magnitude)); // utoa() / ultoa()
165 bufferPos += strlen(str);
166 buffer[bufferPos++] = TAB;
167
168 const auto crc{ calculateChecksum(startPos, bufferPos) };
169 buffer[bufferPos++] = crc;
170
171 buffer[bufferPos++] = CR;
172 }
173
174public:
175 void startFrame()
176 {
177 bufferPos = 0;
178 buffer[bufferPos++] = STX;
179 }
180
181 void send(const char* tag, int16_t value, uint8_t index = 0)
182 {
183 const auto magnitude{ value < 0 ? static_cast< uint16_t >(0U - static_cast< uint16_t >(value)) : static_cast< uint16_t >(value) };
184 writeLine(tag, index, magnitude, value < 0);
185 }
186
187 template< typename Unsigned >
188 void sendUnsigned(const char* tag, Unsigned value)
189 {
190 if constexpr (sizeof(Unsigned) > 2)
191 {
192 writeLine(tag, 0, static_cast< uint32_t >(value), false);
193 }
194 else
195 {
196 writeLine(tag, 0, static_cast< uint16_t >(value), false);
197 }
198 }
199
200 void endFrame()
201 {
202 buffer[bufferPos++] = ETX;
203 }
204
205 [[nodiscard]] const char* data() const
206 {
207 return buffer;
208 }
209 [[nodiscard]] size_t size() const
210 {
211 return bufferPos;
212 }

216template< typename Source >
217static void referenceFrame(Reference& teleInfo)
218{
219 uint8_t idx{ 0 };
220
221 teleInfo.startFrame();
222
223 teleInfo.send("P", Source::power());
224
225 if constexpr (Source::DIVERTED_CT)
226 {
227 teleInfo.send("D", Source::divertedPower());
228 teleInfo.sendUnsigned("E", Source::divertedEnergy());
229 }
230
231 teleInfo.sendUnsigned("V", Source::voltage());
232
233 if constexpr (Source::RELAY_DIVERSION)
234 {
235 teleInfo.send("R", Source::relayAverage());
236
237 idx = Source::RELAY_COUNT;
238 do
239 {
240 --idx;
241 teleInfo.send("R", Source::relayOn(idx), idx + 1);
242 } while (idx);
243 }
244
245 idx = Source::LOAD_COUNT;
246 do
247 {
248 --idx;
249 teleInfo.send("D", Source::loadOnPercent(idx), idx + 1);
250 } while (idx);
251
252 for (uint8_t idx = 0; idx < Source::TEMP_COUNT; ++idx)
253 {
254 if (!Source::temperatureValid(idx))
255 {
256 continue;
257 }
258 teleInfo.send("T", Source::temperature(idx), idx + 1);
259 }
260
261 teleInfo.sendUnsigned("N", Source::absenceCount()); // unsigned since Task 5b, was send() (negative beyond 32767 s)
262
263 if constexpr (Source::DUAL_TARIFF)
264 {
265 teleInfo.send("TA", static_cast< int16_t >(Source::offPeak() ? 1 : 0));
266 }
267
268 teleInfo.sendUnsigned("S", Source::sampleSets());
269 teleInfo.send("S_MC", Source::lowestSampleSetsPerCycle());
270
271 teleInfo.endFrame();
272}

283template< typename Source >
284static Streamed streamFrame()
285{
286 Streamed frame;
287 bool more{ true };
288 while (more)
289 {
290 char step[TeleInfo::MAX_STEP_LENGTH];
291 uint8_t length{ 0 };
292 more = TeleInfo::writeFrameStep< Source >(frame.steps++, step, length);
293 TEST_ASSERT_LESS_OR_EQUAL_UINT8(TeleInfo::MAX_STEP_LENGTH, length);
294 memcpy(frame.text + frame.length, step, length);
295 frame.length += length;
296 if (length > frame.longestStep)
297 {
298 frame.longestStep = length;
299 }
300 TEST_ASSERT_LESS_THAN_UINT8(100, frame.steps); // never endless
301 }
302 return frame;
303}

391/* a frame written independently of the code, checksums computed by hand */
392void test_golden_frame(void)
393{
394 values = Values{};
395 values.power = -25;
396 values.voltage_x100 = 23012;
397 values.loadOnPercent[0] = 40;
398 values.absenceCount = 7;
399 values.sampleSets = 250;
400 values.lowestSampleSetsPerCycle = 44;
401
402 // checksum: the sum of the bytes from the tag to the second TAB, & 0x3F, + 0x20
403 // P\t-25\t 0x0F6 -> 0x36 + 0x20 = 'V'
404 // V\t23012\t 0x160 -> 0x20 + 0x20 = '@'
405 // D1\t40\t 0x0EB -> 0x2B + 0x20 = 'K'
406 // N\t7\t 0x097 -> 0x17 + 0x20 = '7'
407 // S\t250\t 0x0FC -> 0x3C + 0x20 = '\'
408 // S_MC\t44\t 0x1BC -> 0x3C + 0x20 = '\'
409 static const char golden[]{ "\x02"
410 "\nP\t-25\tV\r"
411 "\nV\t23012\t@\r"
412 "\nD1\t40\tK\r"
413 "\nN\t7\t7\r"
414 "\nS\t250\t\\\r"
415 "\nS_MC\t44\t\\\r"
416 "\x03" };
417
418 const Streamed streamed{ streamFrame< Minimal >() };
419 TEST_ASSERT_EQUAL_UINT32(sizeof golden - 1, streamed.length);
420 TEST_ASSERT_EQUAL_MEMORY(golden, streamed.text, sizeof golden - 1);
421}

435static uint8_t parseFrame(const Streamed& frame, Line (&lines)[32])
436{
437 TEST_ASSERT_GREATER_THAN_UINT32(2, frame.length);
438 TEST_ASSERT_EQUAL_CHAR(0x02, frame.text[0]);
439 TEST_ASSERT_EQUAL_CHAR(0x03, frame.text[frame.length - 1]);
440
441 uint8_t count{ 0 };
442 size_t pos{ 1 };
443 while (pos < frame.length - 1)
444 {
445 TEST_ASSERT_EQUAL_CHAR('\n', frame.text[pos++]);
446 const size_t start{ pos };
447 const char* const tab1{ static_cast< const char* >(memchr(frame.text + pos, '\t', frame.length - pos)) };
448 TEST_ASSERT_NOT_NULL(tab1);
449 const char* const tab2{ static_cast< const char* >(memchr(tab1 + 1, '\t', frame.length - (tab1 + 1 - frame.text))) };
450 TEST_ASSERT_NOT_NULL(tab2);
451
452 uint8_t sum{ 0 };
453 for (const char* p = frame.text + start; p != tab2 + 1; ++p)
454 {
455 sum += static_cast< uint8_t >(*p);
456 }
457 TEST_ASSERT_EQUAL_CHAR((sum & 0x3F) + 0x20, tab2[1]);
458 TEST_ASSERT_EQUAL_CHAR('\r', tab2[2]);
459
460 Line& line{ lines[count++] };
461 memset(&line, 0, sizeof line);
462 memcpy(line.tag, frame.text + start, tab1 - (frame.text + start));
463 memcpy(line.value, tab1 + 1, tab2 - (tab1 + 1));
464 pos = (tab2 + 3) - frame.text;
465 }
466 return count;
467}

590/* a single line: LF, tag and index, TAB, value, TAB, checksum, CR */
591void test_line(void)
592{
593 char text[TeleInfo::MAX_STEP_LENGTH];
594
595 // T1\t-5500\t: 0x18E -> 0x0E + 0x20 = '.'

602 TEST_ASSERT_EQUAL_UINT32(11, end - text);
603 TEST_ASSERT_EQUAL_MEMORY("\nE\t65535\t", text, 9);
604
605 // the longest: a 4-character tag and 10 digits
606 end = TeleInfo::lineUnsigned(text, "S_MC", static_cast< uint32_t >(4294967295UL));
607 TEST_ASSERT_EQUAL_UINT32(TeleInfo::lineSize(4, 10), end - text);
608 TEST_ASSERT_EQUAL_UINT8(TeleInfo::MAX_STEP_LENGTH - 1, TeleInfo::lineSize(4, 10));
609
610 // 0 and -32768
611 end = TeleInfo::line(text, "P", 0);
612 TEST_ASSERT_EQUAL_MEMORY("\nP\t0\t", text, 5);
613 end = TeleInfo::line(text, "P", -32768);
614 TEST_ASSERT_EQUAL_MEMORY("\nP\t-32768\t", text, 10);
