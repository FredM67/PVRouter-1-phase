352inline bool printJsonStep(Print& out, uint8_t step)
353{
354 if (step == 0)
355 {
356 out.print(F("{\"P\":"));
357 out.print(tx_data.power);
358
359 if constexpr (RELAY_DIVERSION)
360 {
361 out.print(F(",\"R\":"));
362 out.print(relayDiverter.get_average());
363 }
364 return true;
365 }
366 --step;
367
368 if constexpr (DIVERTED_CT_PRESENT)
369 {
370 if (step == 0)
371 {
372 out.print(F(",\"D\":"));
373 out.print(Shared::copyOf_divertedPower_W);
374 out.print(F(",\"E\":"));
375 out.print(Shared::copyOf_divertedEnergyTotal_Wh);
376 return true;
377 }
378 --step;
379 }
380
381 if constexpr (TEMP_SENSOR_PRESENT)
382 {
383 if (step < tempSensors.size())
384 {
385 if (isValidTemperature(tx_data.temperature_x100[step]))
386 {
387 out.print(F(",\"T"));
388 out.print(step + 1);
389 out.print(F("\":"));
390 SerialOutput::printHundredths(out, tx_data.temperature_x100[step]);
391 }
392 return true;
393 }
394 step -= tempSensors.size();
395 }
396
397 if constexpr (DUAL_TARIFF)
398 {
399 if (step == 0)
400 {
401 out.print(datalogOffPeak ? F(",\"TA\":\"low\"") : F(",\"TA\":\"high\""));
402 return true;
403 }
404 }
405
406 out.println('}');
407 return false;
408}

426inline bool printTextStep(Print& out, uint8_t step)
427{
428 if (step == 0)
429 {
430 SerialOutput::printDecimals(out, Energy::toHundredthsOfJoule< SUPPLY_FREQUENCY >(Shared::copyOf_energyInBucket_main), 2);
431 return true;
432 }
433 if (step == 1)
434 {
435 out.print(F(", P:"));
436 out.print(tx_data.power);
437
438 if constexpr (RELAY_DIVERSION)
439 {
440 out.print(F("/"));
441 out.print(relayDiverter.get_average());
442 }
443 return true;
444 }
445 if (step == 2)
446 {
447 out.print(F(", V:"));
448 SerialOutput::printDecimals(out, tx_data.Vrms_L_x100[0], 2);
449 return true;
450 }
451 step -= 3;
452
453 if constexpr (DIVERTED_CT_PRESENT)
454 {
455 if (step == 0)
456 {
457 out.print(F(", diverted "));
458 out.print(Shared::copyOf_divertedPower_W);
459 out.print(F(" W, "));
460 out.print(Shared::copyOf_divertedEnergyTotal_Wh);
461 out.print(F(" Wh"));
462 return true;
463 }
464 --step;
465 }
466
467 if constexpr (TEMP_SENSOR_PRESENT)
468 {
469 if (step < tempSensors.size())
470 {
471 if (isValidTemperature(tx_data.temperature_x100[step]))
472 {
473 out.print(F(", T"));
474 out.print(step + 1);
475 out.print(F(":"));
476 SerialOutput::printDecimals(out, tx_data.temperature_x100[step], 2);
477 }
478 return true;
479 }
480 step -= tempSensors.size();
481 }
482
483 switch (step)
484 {
485 case 0:
486 out.print(F(", (minSampleSets/MC "));
487 out.print(Shared::copyOf_lowestNoOfSampleSetsPerMainsCycle);
488 return true;
489 case 1:
490 out.print(F(", #ofSampleSets "));
491 out.print(Shared::copyOf_sampleSetsDuringThisDatalogPeriod);
492 return true;
493 case 2:
494 if constexpr (!DUAL_TARIFF && PRIORITY_ROTATION != RotationModes::OFF)
495 {
496 out.print(F(", NoED "));
497 out.print(Shared::absenceOfDivertedEnergyCountInSeconds);
498 }
499 return true;
500 default:
501 out.println(F(")"));
502 return false;
503 }

648inline void sendResults(bool bOffPeak)
649{
650 static bool startup{ true };
651
652 if constexpr (RF_CHIP_PRESENT)
653 {
654 initialize_rf(); // returns at once if the module is up, else retries (about 50 ms when it fails)
655 }
656
657 if (startup)
658 {
659 startup = false;
660 return; // reject the first datalogging which is incomplete !
661 }
662
663 if constexpr (RF_LOGGING_PRESENT)
664 {
665 send_rf_data(tx_data); // *SEND RF DATA*
666 }
667
668 // The output is only started here, and written by SerialOutput::poll() from the
669 // main loop. If the previous one is still in progress (it normally takes a small
670 // part of a datalog period), this one is skipped.
671 if (SerialOutput::busy())
672 {
673 return;
674 }
675
676 if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::HumanReadable)
677 {
678 SerialOutput::start(printTextStep);
679 }
680 else if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::IoT)
681 {
682 datalogOffPeak = bOffPeak;
683 TeleInfoData::snapshot();
684 SerialOutput::start(writeTeleInfoStep);
685 }
686 else if constexpr (SERIAL_OUTPUT_TYPE == SerialOutputType::JSON)
687 {
688 datalogOffPeak = bOffPeak;
689 SerialOutput::start(printJsonStep);
690 }
691}
