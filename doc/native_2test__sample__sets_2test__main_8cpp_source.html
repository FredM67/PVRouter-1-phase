86void test_datalog_sums_hold_30_s(void)
87{
88 // each sample adds at most 2^18 (|V_ADC x I_ADC| or V_ADC^2); for long periods both 32-bit sums
89 // are scaled down by 16 (Datalog::powerShift(), Datalog::isVSquaredScaled()), the power rounded
90 // once per cycle (+ 1 per cycle at most). Checked in 64 bits, here only.
91 constexpr uint64_t fullScale{ 1ULL << 18 };
92 constexpr uint64_t at30s{ SampleSets::maxPerDatalog(30 * 60, N_MAX_60HZ) }; // the larger of 50/60 Hz
93
94 TEST_ASSERT_TRUE(at30s * (fullScale >> 4) <= UINT32_MAX);
95 TEST_ASSERT_TRUE(at30s * (fullScale >> 4) + 30 * 60 <= static_cast< uint64_t >(INT32_MAX));
96}
