77constexpr PinList pinsOf(const DisplayType type)
78{
79 PinList pins{};

254constexpr PortImage charImageOf(const DisplayType type, const uint8_t character)
255{
256 PortImage image{};
257 const uint8_t c{ character < NO_OF_CHARS ? character : CHAR_BLANK };
258
259 if (type == DisplayType::SEG)
260 {
261 for (uint8_t segment = 0; segment != 7; ++segment)
262 {
263 if (segMap[c][segment])
264 {
265 image |= imageOf(segmentDrivePin[segment]);
266 }
267 }
268 }
269 else if (type == DisplayType::SEG_HW)
270 {
271 for (uint8_t line = 0; line != 4; ++line)
272 {
273 if (digitValueMap[c][line])
274 {
275 image |= imageOf(digitSelectionLine[line]);
276 }
277 }
278 }
279 return image;
280}

297constexpr PortImage digitImageOf(const DisplayType type, const uint8_t digit)
298{
299 PortImage image{};
300 if (type == DisplayType::SEG)
301 {
302 for (uint8_t i = 0; i != NO_OF_DIGITS; ++i)
303 {
304 if (i != digit)
305 {
306 image |= imageOf(digitSelectorPin[i]);
307 }
308 }
309 }
310 else if (type == DisplayType::SEG_HW)
311 {
312 for (uint8_t line = 0; line != 2; ++line)
313 {
314 if (digitLocationMap[digit & 3][line])
315 {
316 image |= imageOf(digitLocationLine[line]);
317 }
318 }
319 }
320 return image;
321}

327constexpr PortImage imageOf(const DisplayType type, const uint8_t code, const uint8_t digit)
328{
329 PortImage image{ charImageOf(type, static_cast< uint8_t >(code & 0x7F)) };
330 if (code & DP)
331 {
332 image |= dpImageOf(type);
333 }
334 image |= digitImageOf(type, digit);
335 return image;

401constexpr Item itemAt(const uint8_t step, const bool hasSlot, const Content slot, const uint8_t noOfSensors)
402{
403 const uint8_t noOfItems{ static_cast< uint8_t >(noOfSensors + (hasSlot ? 1 : 0)) };
404 if (noOfItems == 0)
405 {
406 return { Content::WAITING, 0 };
407 }
408 uint8_t index{ static_cast< uint8_t >(step % noOfItems) };
409 if (hasSlot)
410 {
411 if (index == 0)
412 {
413 return { slot, 0 };
414 }
415 --index;
416 }
417 return { Content::TEMPERATURE, index };
418}

454constexpr void segEnergy(const uint16_t wh, uint8_t (&chars)[NO_OF_DIGITS])
455{
456 const bool above10kWh{ wh > 9999U };
457 uint16_t value{ above10kWh ? static_cast< uint16_t >(wh / 10U) : wh };
458
459 for (uint8_t i = NO_OF_DIGITS; i != 0; --i)
460 {
461 chars[i - 1] = static_cast< uint8_t >(value % 10U);
462 value /= 10U;
463 }
464 chars[above10kWh ? 1 : 0] |= DP;
465}

478constexpr void segTemperature(const int16_t temperature_x100, const bool letters, uint8_t (&chars)[NO_OF_DIGITS])
479{
480 const uint8_t unit{ letters ? CHAR_C : CHAR_BLANK };
481 const uint8_t minus{ letters ? CHAR_MINUS : static_cast< uint8_t >(CHAR_BLANK | DP) };
482
483 if (!isValidTemperature(temperature_x100))
484 {
485 for (auto& c : chars)
486 {
487 c = letters ? CHAR_MINUS : CHAR_BLANK;
488 }
489 return;
490 }
491
492 const uint16_t magnitude{ static_cast< uint16_t >(temperature_x100 < 0 ? -static_cast< int32_t >(temperature_x100) : temperature_x100) };
493 const uint16_t tenths{ static_cast< uint16_t >((magnitude + 5U) / 10U) }; // 0.1 °C, rounded
494 const bool negative{ (temperature_x100 < 0) && (tenths != 0) }; // no "-0.0"
495
496 chars[3] = unit;
497 if (tenths < 100U)
498 {
499 // -9.9 to 9.9 °C: [sign or blank][units.][tenths]
500 chars[2] = static_cast< uint8_t >(tenths % 10U);
501 chars[1] = static_cast< uint8_t >(((tenths / 10U) % 10U) | DP);
502 chars[0] = negative ? minus : CHAR_BLANK;
503 }
504 else if (!negative)
505 {
506 // [tens][units.][tenths] up to 99.9 °C, then [hundreds][tens][units]
507 if (tenths < 1000U)
508 {
509 chars[2] = static_cast< uint8_t >(tenths % 10U);
510 chars[1] = static_cast< uint8_t >(((tenths / 10U) % 10U) | DP);
511 chars[0] = static_cast< uint8_t >(tenths / 100U);
512 }
513 else
514 {
515 const uint16_t degrees{ static_cast< uint16_t >((tenths + 5U) / 10U) };
516 chars[2] = static_cast< uint8_t >(degrees % 10U);
517 chars[1] = static_cast< uint8_t >((degrees / 10U) % 10U);
518 chars[0] = static_cast< uint8_t >((degrees / 100U) % 10U);
519 }
520 }
521 else
522 {
523 // below -9.9 °C: [-][tens][units]
524 const uint16_t degrees{ static_cast< uint16_t >((tenths + 5U) / 10U) };
525 chars[2] = static_cast< uint8_t >(degrees % 10U);
526 chars[1] = static_cast< uint8_t >((degrees / 10U) % 10U);
527 chars[0] = minus;
528 }
529}

537constexpr void oledEnergy(const uint16_t wh, char (&text)[OLED_TEXT_SIZE])
538{
539 uint8_t chars[NO_OF_DIGITS]{};
540 segEnergy(wh, chars);
541
542 uint8_t pos{ 0 };
543 for (const auto c : chars)
544 {
545 text[pos++] = static_cast< char >('0' + (c & 0x7F));
546 if (c & DP)

548 text[pos++] = '.';
549 }
550 }
551 text[pos] = '\0';

558constexpr void oledTemperature(const int16_t temperature_x100, char (&text)[OLED_TEXT_SIZE])
559{
560 for (auto& c : text)
561 {
562 c = ' ';
563 }
564 text[OLED_TEXT_SIZE - 1] = '\0';
565
566 if (!isValidTemperature(temperature_x100))
567 {
568 text[1] = '-';
569 text[2] = '-';
570 text[3] = '.';
571 text[4] = '-';
572 return;
573 }
574
575 const uint16_t magnitude{ static_cast< uint16_t >(temperature_x100 < 0 ? -static_cast< int32_t >(temperature_x100) : temperature_x100) };
576 uint16_t tenths{ static_cast< uint16_t >((magnitude + 5U) / 10U) }; // 0.1 °C, rounded
577 const bool negative{ (temperature_x100 < 0) && (tenths != 0) }; // no "-0.0"
578 if (tenths > 9999U)
579 {
580 tenths = 9999U; // the readings stay far below (DS18B20: 125 °C)
581 }
582
583 int8_t pos{ OLED_TEXT_SIZE - 2 };
584 text[pos--] = static_cast< char >('0' + tenths % 10U);
585 text[pos--] = '.';
586 tenths /= 10U;
587 do
588 {
589 text[pos--] = static_cast< char >('0' + tenths % 10U);
590 tenths /= 10U;
591 } while ((tenths != 0) && (pos >= 0));
592
593 if (negative && (pos >= 0))
594 {
595 text[pos] = '-';
596 }
597}
