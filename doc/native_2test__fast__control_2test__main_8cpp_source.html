59template< uint8_t NMin, uint8_t NMax, uint8_t SumShift >
60static void checkDivisionFree(const float cal, const uint8_t samplesPerCycle)
61{
62 const float cals[1]{ cal };
63 const auto table{ Energy::toFixedPerSample< NMin, NMax, SumShift >(cals) };
64
65 static constexpr int32_t sums[]{ 0, 1, -1, 1000, -1000, 123456, -123456, 2000000, -2000000, (34L << 18) - 1, -((34L << 18) - 1) };
66
67 uint8_t checked{ 0 };
68 for (uint8_t n = 1; n != NMax; ++n)
69 {
70 const uint8_t index{ FastControl::wholeCycleIndexOf(n, NMin, NMax) };
71 if ((2 * n < NMin) || (2 * n > NMax))
72 {
73 TEST_ASSERT_EQUAL_UINT8(FastControl::NO_INDEX, index);
74 continue;
75 }
76 TEST_ASSERT_EQUAL_UINT8(2 * n - NMin, index);
77 ++checked;
78
79 for (const int32_t sum : sums)
80 {
81 if ((sum > (static_cast< int32_t >(n) << 18)) || (sum < -(static_cast< int32_t >(n) << 18)))
82 {
83 continue; // beyond full scale for n sample sets
84 }
85 const int32_t wholeCycle{ Energy::contribution(sum >> SumShift, table.value[0][samplesPerCycle - NMin], table.shift) };
86 const int32_t expected{ FastControl::predictEndOfCycle(1000, wholeCycle, n, samplesPerCycle) };
87 const int32_t actual{ FastControl::predictEndOfCycle(1000, sum, table.value[0][index], SumShift, table.shift) };
88
89 const int32_t magnitude{ expected > 1000 ? expected - 1000 : 1000 - expected };
90 TEST_ASSERT_INT32_WITHIN(2 + magnitude / 8192, expected, actual);
91 }
92 }
93 TEST_ASSERT_TRUE(checked >= 4);
94}

125void test_decision_uses_prediction_only()
126{
127 constexpr LoadPriorities::Bucket bucket{ 1000, 500, 500, 500 };
128 LoadPriorities::Switching< 2 > switching;
129 uint8_t priorities[2]{ 0, 1 };
130
131 const int32_t measured{ 400 }; // the bucket at the start of the cycle: below the middle
132 // the first half brings 250 units over 32 of 64 sample sets: 500 over the cycle
133 const int32_t prediction{ FastControl::predictEndOfCycle(measured, 250, 32, 64) };
134 TEST_ASSERT_EQUAL_INT32(900, prediction);
135
136 switching.onNewCycle(priorities, prediction, bucket);
137
138 // Upper half: load 0 ON, the lower threshold reset to its default, the upper one untouched
139 // (still its initial 0). Fed the measured 400, load 0 would stay OFF, the upper threshold would
140 // be reset to 500 and the lower one would stay 0.
141 TEST_ASSERT_TRUE((priorities[0] & loadStateOnBit) != 0);
142 TEST_ASSERT_TRUE((priorities[1] & loadStateOnBit) == 0);
143 TEST_ASSERT_EQUAL_INT32(500, switching.lowerThreshold);
144 TEST_ASSERT_EQUAL_INT32(0, switching.upperThreshold);
145}
