28struct Robin
29{
30 int32_t lpf_long{};
31
32 int32_t apply(const int16_t countsMinusDC, const float lpf_gain, const float alpha)
33 {
34 int32_t sampleIminusDC_grid = ((int32_t)countsMinusDC) << 8;
35 const int32_t last_lpf_long = lpf_long;
36 lpf_long += alpha * (sampleIminusDC_grid - last_lpf_long);
37 sampleIminusDC_grid += (lpf_gain * lpf_long);
38 return sampleIminusDC_grid;
39 }
40};

98void test_time_constant_within_2_percent_of_robins(void)
99{
100 // Robin's low-pass, in float: lpf += alpha x (x - lpf). After a step, it reaches 1 - 1/e of
101 // the step in -1 / ln(1 - alpha) samples (908.6 with alpha 0.0011).
102 const float expected{ static_cast< float >(-1.0 / log(1.0 - static_cast< double >(ALPHA))) };
103
104 static constexpr int16_t STEP{ 200 }; // counts
105 CtFilter::LowPassBoost filter;
106 uint16_t samples{ 0 };
107 while (lowPassCounts(filter) < static_cast< float >(STEP * (1.0 - exp(-1.0))))
108 {
109 filter.apply(x64(STEP), COEFFS);
110 ++samples;
111 TEST_ASSERT_TRUE(samples < 10000);
112 }
113
114 TEST_ASSERT_FLOAT_WITHIN(expected * 0.02F, expected, static_cast< float >(samples));
115}

117void test_steady_state_within_one_count_of_robins(void)
118{
119 // a 50 Hz current of 200 counts peak with a DC residual of 5 counts, 64 samples per cycle
120 static constexpr int16_t sine64[16]{ 0, 20, 39, 56, 71, 83, 92, 98, 100, 98, 92, 83, 71, 56, 39, 20 };
121
122 CtFilter::LowPassBoost filter;
123 Robin robin;
124
125 for (uint16_t i = 0; i != 64U * 50U; ++i) // one second
126 {
127 const uint8_t k{ static_cast< uint8_t >(i % 64) };
128 const int16_t quarter{ sine64[k % 16] };
129 const int16_t counts{ static_cast< int16_t >(5 + 2 * ((k < 32) ? quarter : -quarter)) };
130
131 const int32_t expected{ robin.apply(counts, GAIN, ALPHA) >> 2 }; // back to x64
132 const int16_t actual{ filter.apply(x64(counts), COEFFS) };
133
134 TEST_ASSERT_INT32_WITHIN(64, expected, actual); // one ADC count
135 }
136
137 // and the state has settled on the DC residual as Robin's has
138 TEST_ASSERT_FLOAT_WITHIN(1.0F, static_cast< float >(robin.lpf_long) / 256.0F, lowPassCounts(filter));
139}
