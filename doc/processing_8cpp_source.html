371void initializeProcessing()
372{
373 // Initialize DC offset in Q16.16 format: (0x8000 << 16) for left-aligned ADC
374 // 0x8000 is the mid-point of a left-aligned 10-bit ADC result
375 initializeArray(i_DCoffset_V, i_DCoffset_V_nom); // nominal mid-point value of ADC left-aligned, x64 scale
376
377 constexpr uint32_t filterDC_V_nom = static_cast< uint32_t >(i_DCoffset_V_nom) << 15;
378 initializeArray(l_filterDC_V, filterDC_V_nom); // nominal mid-point value of ADC left-aligned
379
380 setPinsAsOutput< getOutputPins() >(); // set the output pins as OUTPUT
381 setPinsAsInputPullup< getInputPins() >(); // set the input pins as INPUT_PULLUP
382
383 uint8_t i{ NO_OF_DUMPLOADS };
384 do
385 {
386 --i;
387 loadPrioritiesAndState[i] = loadPrioritiesAtStartup[i];
388 loadPrioritiesAndState[i] &= loadStateMask;
389 } while (i);
390
391 if constexpr (RF_CHIP_PRESENT)
392 {
393 // Initialize shared RF module
394 if (initialize_rf())
395 {
396 infoln(F("RF module initialized"));
397 }
398 else
399 {
400 infoln(F("RF module initialization FAILED"));
401 }
402 }
403
404 if constexpr (REMOTE_LOADS_PRESENT)
405 {
406 remoteLoads.reset();
407 infoln(F("Remote loads initialized"));
408 }
409
410 // The ADC runs free: each conversion starts the next one, and each end of conversion
411 // raises the ADC interrupt, where the samples are processed.
412 ADCSRA = 0; // ADC off while it is set up
413 ADCSRB = 0; // ADTS2:0 = 0: the auto-trigger source is the ADC itself (free running)
414
415 // Reference and left-aligned result for the first conversion too, on the first channel (V)
416 ADMUX = AdcChain::admuxOf(PCB_VERSION, AdcChain::Channel::V);
417
418 ADCSRA = bit(ADEN) // ADC on
419 | bit(ADSC) // first conversion, the next ones start by themselves
420 | bit(ADATE) // auto-trigger, from the source set in ADCSRB
421 | bit(ADIE) // interrupt at the end of each conversion
422 | bit(ADPS2) | bit(ADPS1) | bit(ADPS0); // ADC clock 16 MHz / 128 = 125 kHz: 104 µs per conversion
423
424 sei(); // Enable Global Interrupts
425}

467void updatePortsStates()
468{
469 uint16_t pinsON{ 0 };
470 uint16_t pinsOFF{ 0 };
471
472 // One pass over the load map. A local load drives its TRIAC pin; a remote load is
473 // driven over RF and only drives its optional status LED here, if it has one.
474 uint8_t i{ NO_OF_DUMPLOADS };
475 do
476 {
477 --i;
478
479 const uint8_t loadEntry{ physicalLoadPin[i] };
480 // with an all-local map both of these fold away at compile time, leaving the original code
481 const uint8_t pin{ REMOTE_LOADS_PRESENT ? Load::pinOf(loadEntry) : loadEntry };
482 const bool bDrivesAPin{ !REMOTE_LOADS_PRESENT || Load::isLocal(loadEntry) || (pin != 0) };
483
484 if (LoadStates::LOAD_OFF == physicalLoadState[i])
485 {
486 if (bDrivesAPin)
487 {
488 pinsOFF |= bit(pin);
489 }
490 }
491 else
492 {
493 // count ON-cycles on the load's *state*, so remote loads are datalogged too
494 ++countLoadON[i];
495
496 if (bDrivesAPin)
497 {
498 pinsON |= bit(pin);
499 }
500 }
501 } while (i);
502
503 setPinsState< getLoadPins() >(pinsON, pinsOFF);
504}

613void processCurrentRawSample(const uint8_t phase, const uint16_t rawSample)
614{
615 // extra items for an LPF to improve the processing of data samples from CT1
616 static CtFilter::LowPassBoost ctFilter[NO_OF_PHASES]; // for offsetting the behaviour of CTx as a HPF
617
618 // remove most of the DC offset from the current sample.
619 // Use the voltage bias, it's most likely more accurate than the default mid-range.
620 // Add 0.5 for the rounding, +32 when 10-bits left aligned `1<<(15-10)`
621 int16_t sampleIminusDC = (rawSample | 32U) - i_DCoffset_V[phase];
622
623 // extra filtering to offset the HPF effect of CTx
624 // Using if constexpr ensures zero overhead when CT filtering is disabled
625 if constexpr (lpf_gain != 0.0F && alpha != 0.0F)
626 {
627 static_assert(CtFilter::isAccurate(lpf_gain, alpha), "******** CT1 filter: alpha must be within 0.00005..0.0625 and lpf_gain below 64 (calibration.h) ! ********");
628 constexpr CtFilter::Coefficients ctFilterCoefficients{ CtFilter::coefficientsOf(lpf_gain, alpha) };
629
630 sampleIminusDC = ctFilter[phase].apply(sampleIminusDC, ctFilterCoefficients); // Robin's filter, at x256, in fixed point, saturated
631 }
632
633 // calculate the "real power" in this sample pair and add to the accumulated sum
634 // Reduce to 14-bits (x16 scale) before multiplication for efficiency on 8-bit MCU
635 const int16_t filtV_div4{ i_sampleVminusDC[phase] >> 2 }; // reduce to 14-bits (now x16, or 2^4)
636 const int16_t filtI_div4{ sampleIminusDC >> 2 }; // reduce to 14-bits (now x16, or 2^4)
637
638 // Using optimized assembly multiplication for ~3x speedup in ISR
639 int32_t instP;
640 multS16x16_to32(instP, filtV_div4, filtI_div4); // 32-bits (now x256, or 2^8)
641 instP >>= 8; // scaling is now x1, as for Mk2 (V_ADC x I_ADC)
642
643 l_sumP[phase] += instP; // cumulative power, scaling as for Mk2 (V_ADC x I_ADC); into the datalog sum once per cycle
644}

733void processVoltage(const uint8_t phase)
734{
735 // for the Vrms calculation (for datalogging only)
736 // Reduce to 14-bits (x16 scale) before multiplication for efficiency on 8-bit MCU
737 const int16_t filtV_div4{ i_sampleVminusDC[phase] >> 2 }; // reduce to 14-bits (now x16, or 2^4)

742 uint32_t inst_Vsquared = static_cast< uint32_t >(inst_Vsquared_signed);
743
744 // The sum cannot overflow: checked at compile time with the sample count (top of this file).
745 // Datalog::vrmsFactor() compensates the x1/16 scaling, on the same condition.
746 if constexpr (Datalog::isVSquaredScaled(DATALOG_PERIOD_IN_SECONDS))
747 {
748 inst_Vsquared >>= 12; // scaling is now x1/16 (V_ADC x V_ADC)
749 }
750 else
751 {
752 inst_Vsquared >>= 8; // scaling is now x1 (V_ADC x V_ADC)
753 }
754
755 l_sum_Vsquared[phase] += inst_Vsquared; // cumulative V^2 (V_ADC x V_ADC)
756 //
757 // store items for use during next loop
758 l_filterDC_V[phase] += i_sampleVminusDC[phase]; // for use with LP filter
759 polarityConfirmedOfLastSampleV[phase] = polarityConfirmed[phase]; // for identification of half cycle boundaries
760 ++n_samplesDuringThisMainsCycle[phase]; // for real power calculations
761}

804//
805// The ADC interrupt writes ADMUX last, and that write must land within one conversion (1664
806// cycles) of the interrupt, even when the interrupt waited for the end of a 1-Wire time slot
807// (interrupts off up to ~1150 cycles, temperature sensing): every call must reach its ADMUX

809// must be taken on that very sample, and posts the rest of the work as tasks; each CT2 call,

990void processLatestContribution()
991{
992 // for efficiency, the energy scale is Joules * SUPPLY_FREQUENCY, in integer fixed point
993 // add the latest energy contribution to the main energy accumulator
994 // (sumP / n) x cal == sumP x (cal / n): cal / n comes from the table, so there is no division
995 const uint8_t n{ n_samplesLastCycle };
996 int32_t sumP{ l_sumP_lastCycle };
997 uint8_t index{ static_cast< uint8_t >(n - N_MIN) };
998
999 // the new cycle's samples so far stay in the sum (processPlusHalfCycle())
1000 l_sumP[0] -= sumP;
1001
1002 if ((n < N_MIN) || (n > N_MAX))
1003 {
1004 // unexpected cycle length (start-up, missing phase): rescale the sum to N_MIN samples
1005 sumP = rescaleToNMin(sumP, n);
1006 index = 0;
1007 }
1008
1009 l_energyInBucket_main += Energy::contribution(sumP >> SUM_P_SHIFT, pgm_read_word(&powerCalPerSample.value[0][index]), powerCalPerSampleShift);
1010
1011 // apply any adjustment that is required.
1012 // If diversion hasn't started yet, use start threshold, otherwise use regular offset
1013 if (!b_diversionStarted)
1014 {
1015 l_energyInBucket_main -= Energy::fromWatts(DIVERSION_START_THRESHOLD_WATTS);
1016
1017 // Check if we've exceeded the threshold to start diversion
1018 if (l_energyInBucket_main > l_upperThreshold_default)
1019 {
1020 b_diversionStarted = true;
1021 // Once started, we divert all surplus according to the configured fixed offset
1022 }
1023 }
1024 else
1025 {
1026 // When diversion is already started, apply normal export offset if configured
1027 // Comment or remove this if you want to divert ALL surplus once started
1028 l_energyInBucket_main -= Energy::fromWatts(REQUIRED_EXPORT_IN_WATTS);
1029 }
1030
1031 // Min and max limits to the level of the energy bucket, once the cycle's contribution is in
1032 // (the decision of this cycle comes later, at mid-cycle, on the prediction). This is to
1033 // ensure correct operation when conditions change, i.e. when import changes to export, and
1034 // vice versa.
1035 if (l_energyInBucket_main > l_capacityOfEnergyBucket_main)
1036 {
1037 l_energyInBucket_main = l_capacityOfEnergyBucket_main;
1038 }
1039 else if (l_energyInBucket_main < 0)
1040 {
1041 l_energyInBucket_main = 0;
1042 }
1043}

1352void processRawSamples(const uint8_t phase)
1353{
1354 // The raw V and I samples are processed in "phase pairs"
1355 const auto &lastPolarity{ polarityConfirmedOfLastSampleV[phase] };
1356
1357 if (Polarities::POSITIVE == polarityConfirmed[phase])
1358 {
1359 // the polarity of this sample is positive
1360 if (Polarities::POSITIVE != lastPolarity)
1361 {
1362 // This is the start of a new +ve half cycle, for this phase, just after the zero-crossing point.
1363 if (beyondStartUpPeriod)
1364 {
1365 processPlusHalfCycle(phase);
1366 }
1367 else
1368 {
1369 processStartUp(phase);
1370 }
1371 }
1372 }
1373 else
1374 {
1375 // the polarity of this sample is negative
1376 if (Polarities::NEGATIVE != lastPolarity)
1377 {
1378 // This is the start of a new -ve half cycle (just after the zero-crossing point)
1379 processMinusHalfCycle(phase);
1380 if (0 == phase)
1381 {
1382 n_samplesDuringNegativeHalf = 0;
1383 }
1384 }
1385
1386 // The loads are decided at mid-cycle, on the prediction (fast_control.h): the voltage is now
1387 // far enough past the zero crossing for the trigger devices to be reliably armed, and they
1388 // latch the new states at the next positive zero crossing. The count stops past the decision,
1389 // so that a negative voltage that lasts never wraps it into a second decision.
1390 if ((0 == phase) && (n_samplesDuringNegativeHalf <= FastControl::DECISION_SAMPLE))
1391 {
1392 if (beyondStartUpPeriod && (FastControl::DECISION_SAMPLE == n_samplesDuringNegativeHalf))
1393 {
1394 // Once per mains cycle, at mid-cycle on phase 0: the decision and the ports, on the
1395 // next two CT2 calls (CycleTask)
1396 pendingCycleTasks |= CycleTask::DECISION;
1397 }
1398 ++n_samplesDuringNegativeHalf;
1399 }
1400 }
