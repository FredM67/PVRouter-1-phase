71{

75 {
76 digits[count++] = static_cast< char >('0' + value % 10);

101template< typename Magnitude >
102char* writeLine(char* dst, const char* tag, uint8_t index, Magnitude magnitude, bool negative)
103{
104 *dst++ = LF;
105
106 char* const start{ dst };
107 while (*tag)
108 {
109 *dst++ = *tag++;
110 }
111 if (index != 0)
112 {
113 *dst++ = static_cast< char >('0' + index);
114 }
115 *dst++ = TAB;
116
117 if (negative)
118 {
119 *dst++ = '-';
120 }
121 dst = writeDigits(dst, magnitude);
122 *dst++ = TAB;
123

125 for (const char* ptr = start; ptr != dst; ++ptr)
126 {
127 sum += static_cast< uint8_t >(*ptr);
128 }
129 *dst++ = static_cast< char >((sum & 0x3F) + 0x20);
130
131 *dst++ = CR;
132 return dst;
133}

178{

208template< typename Source >
209char* frameLine(uint8_t step, char* dst)
210{

212 {
213 return line(dst, "P", Source::power());
214 }
215 --step;
216
217 if constexpr (Source::DIVERTED_CT)
218 {
219 if (step == 0)
220 {
221 return line(dst, "D", Source::divertedPower());
222 }
223 if (step == 1)
224 {
225 return lineUnsigned(dst, "E", Source::divertedEnergy());
226 }
227 step -= 2;
228 }
229
230 if (step == 0)
231 {
232 return lineUnsigned(dst, "V", Source::voltage());
233 }
234 --step;
235
236 if constexpr (Source::RELAY_DIVERSION)
237 {
238 if (step == 0)
239 {
240 return line(dst, "R", Source::relayAverage());
241 }
242 --step;
243
244 if (step < Source::RELAY_COUNT) // Rn first
245 {
246 const uint8_t index{ static_cast< uint8_t >(Source::RELAY_COUNT - step) };
247 return line(dst, "R", Source::relayOn(index - 1), index);
248 }
249 step -= Source::RELAY_COUNT;
250 }
251
252 if (step < Source::LOAD_COUNT) // Dn first
253 {
254 const uint8_t index{ static_cast< uint8_t >(Source::LOAD_COUNT - step) };
255 return line(dst, "D", Source::loadOnPercent(index - 1), index);
256 }
257 step -= Source::LOAD_COUNT;
258
259 if constexpr (Source::TEMP_COUNT != 0)
260 {
261 if (step < Source::TEMP_COUNT)
262 {
263 return Source::temperatureValid(step) ? line(dst, "T", Source::temperature(step), step + 1) : dst;
264 }
265 step -= Source::TEMP_COUNT;
266 }
267
268 if (step == 0)

270 return lineUnsigned(dst, "N", Source::absenceCount());

276 if (step == 0)

278 return line(dst, "TA", Source::offPeak() ? 1 : 0);

281 }
282

326{
327 char* end{ text };
