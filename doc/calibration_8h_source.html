17//
18// Calibration values
19//-------------------
20// Each board has its own set below: the router uses the one of PCB_VERSION (config.h).
21// f_powerCal converts the products of voltage and current samples into watts: CT1 (grid) for the
22// power the router diverts, CT2 (diverted) for the diverted energy it reports. f_voltageCal
23// converts the voltage samples into volts, for the voltage shown in the datalog only. To
24// calibrate, compare with a reference meter:
25// new value = old value x reference reading / router reading
26//
27// ---- Old boards (PcbVersion::OLD_SEG and PcbVersion::OLD_OLED) ----
28//
29// When calculating real power, which is what this code does, the individual
30// conversion rates for voltage and current are not of importance. It is
31// only the conversion rate for POWER which is important. This is the
32// product of the individual conversion rates for voltage and current. It
33// therefore has the units of ADC-steps squared per Watt. Most systems will
34// have a power conversion rate of around 20 (ADC-steps squared per Watt).
35//
36// powerCal is the RECIPROCAL of the power conversion rate. A good value
37// to start with is therefore 1/20 = 0.05 (Watts per ADC-step squared)
38//
39inline constexpr float f_powerCalOldBoard{ 0.05000F };
40inline constexpr float f_powerCalDivertedOldBoard{ 0.05000F };
41//
42// For datalogging purposes, f_voltageCal has been added too. Because the range of ADC values is
43// similar to the actual range of volts, the optimal value for this cal factor is likely to be
44// close to unity.
45inline constexpr float f_voltageCalOldBoard{ 0.805653F };
46//
47// CT1 filter: an IIR low-pass filter of the grid current, added back with a gain, offsets the CT
48// acting as a high-pass filter. Interim values for the SCT-013-000 (Dec. 2024, on real hardware).
49// Turned into fixed point at compile time (ct_filter.h): alpha from 0.00005 to 0.0625, gain below 64.
50inline constexpr float lpf_gainOldBoard{ 9 };
51inline constexpr float alphaOldBoard{ 0.0011 };
52//
53// ---- New board (PcbVersion::NEW, rev. 6.0+) ----
54//
55// f_powerCal is given per amp of CT rating: f_powerCal = f_powerCalPerAmp x f_ctRating, so
56// changing a CT only means entering its rating. The default values suit 333 mV voltage-output CTs;
57// they come from the design of the board, corrected by the calibration of a first board (see
58// docs/calibration.md). With a current-output CT and a burden resistor R (YHDC SCT-013-000,
59// 2000:1), f_ctRating = 666 / R: the current giving 333 mV across the burden.
60inline constexpr float f_ctRating{ 50.0F };
61inline constexpr float f_ctRatingDiverted{ 50.0F };
62inline constexpr float f_powerCalPerAmp{ 0.00286405F };
63inline constexpr float f_voltageCalNewBoard{ 0.888065256F };
64//
65// CT1 filter: off, the voltage-output CTs behave differently.
66inline constexpr float lpf_gainNewBoard{ 0 };
67inline constexpr float alphaNewBoard{ 0.002 };
68//
69// ---- All boards ----
70//
71// f_phaseCal is used to alter the phase of the voltage waveform relative to the current waveform.
72// The algorithm interpolates between the most recent pair of voltage samples according to the value of f_phaseCal.
73//
74// With f_phaseCal = 1, the most recent sample is used.
75// With f_phaseCal = 0, the previous sample is used
76// With f_phaseCal = 0.5, the mid-point (average) value in used
77//
78// NB. Any tool which determines the optimal value of f_phaseCal must have a similar
79// scheme for taking sample values as does this sketch.
80//
81inline constexpr float f_phaseCal{ 1 };
