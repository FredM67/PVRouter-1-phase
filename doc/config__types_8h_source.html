166template< uint8_t MaxPins >
167struct PinList
168{
169 uint8_t pins[MaxPins];
170 uint8_t count;
171
175 constexpr PinList()
176 : pins{}, count(0) {}
177
182 constexpr PinList(uint16_t bitmask)
183 : pins{}, count(0)
184 {
185 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
186 {
187 if (bitmask & (1U << pin))
188 {
189 pins[count++] = pin; // Store the physical pin number
190 }
191 }
192 }
193
200 constexpr PinList(uint32_t bitmask)
201 : pins{}, count(0)
202 {
203 // Extract local pins from lower 16 bits
204 for (uint8_t pin = 0; pin < 16 && count < MaxPins; ++pin)
205 {
206 if (bitmask & (1UL << pin))
207 {
208 pins[count++] = pin; // Store the physical pin number
209 }
210 }
211 // Extract remote loads from upper 16 bits
212 for (uint8_t i = 0; i < 8 && count < MaxPins; ++i)
213 {
214 if (bitmask & (1UL << (16 + i)))
215 {
216 pins[count++] = REMOTE_PIN_BASE + i; // Store virtual pin for remote load
217 }
218 }
219 }
220
225 template< typename... Args >
226 constexpr PinList(Args... args)
227 : pins{ static_cast< uint8_t >(args)... }, count(sizeof...(args)) {}
228
233 constexpr uint16_t toLocalBitmask() const
234 {
235 uint16_t result{ 0 };
236 for (uint8_t i = 0; i < count; ++i)
237 {
238 if (pins[i] < REMOTE_PIN_BASE)
239 {
240 result |= (1U << pins[i]);
241 }
242 }
243 return result;
244 }
245
251 constexpr uint8_t toRemoteBitmask() const
252 {
253 uint8_t result{ 0 };
254 for (uint8_t i = 0; i < count; ++i)
255 {
256 if (pins[i] >= REMOTE_PIN_BASE)
257 {
258 const uint8_t remoteIndex = pins[i] - REMOTE_PIN_BASE;
259 if (remoteIndex < REMOTE_MASK_BITS)
260 {
261 result |= (1U << remoteIndex);
262 }
263 }
264 }
265 return result;
266 }
267};
