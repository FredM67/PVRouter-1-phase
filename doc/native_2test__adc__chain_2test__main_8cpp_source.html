31static Channel channelOfAdmux(const PcbVersion board, const uint8_t admux)
32{
33 const uint8_t pin{ static_cast< uint8_t >(admux & MUX_MASK) };
34 constexpr Channel channels[]{ Channel::V, Channel::I_GRID, Channel::I_DIVERTED };
35 for (const Channel ch : channels)
36 {
37 if (AdcChain::pinOf(board, ch) == pin)
38 {
39 return ch;
40 }
41 }
42 TEST_FAIL_MESSAGE("ADMUX selects an input that is no channel of the board");
43 return Channel::V;
44}

72void test_pipeline_order(void)
73{
74 constexpr uint8_t ROUNDS{ 30 };
75
76 for (const PcbVersion board : boards)
77 {
78 // admuxOf[k]: the ADMUX the conversion of round k runs with
79 uint8_t admuxOf[ROUNDS + 2]{};
80
81 // the chain primed with V, then I_GRID
82 admuxOf[0] = AdcChain::refBits(board) | ADLAR_BIT | AdcChain::pinOf(board, Channel::V);
83 admuxOf[1] = AdcChain::refBits(board) | ADLAR_BIT | AdcChain::pinOf(board, Channel::I_GRID);
84
85 for (uint8_t round = 0; round != ROUNDS; ++round)
86 {
87 // the ISR of round k believes it handles channelConverted(k)...
88 const Channel believed{ AdcChain::channelConverted(round) };
89 TEST_ASSERT_EQUAL_UINT8(round % 3, static_cast< uint8_t >(believed)); // V, I_GRID, I_DIVERTED, V...
90
91 // ... and the ADC really converted the channel its ADMUX selected
92 TEST_ASSERT_EQUAL_UINT8(static_cast< uint8_t >(believed), static_cast< uint8_t >(channelOfAdmux(board, admuxOf[round])));
93 TEST_ASSERT_EQUAL_HEX8(AdcChain::refBits(board), admuxOf[round] & REF_MASK);
94 TEST_ASSERT_EQUAL_HEX8(ADLAR_BIT, admuxOf[round] & ADLAR_BIT);
95
96 // the conversion of round k+1 has already started: the write of round k applies to round k+2
97 admuxOf[round + 2] = AdcChain::admuxWrittenAfter(board, believed);
98 }
99 }
100}
