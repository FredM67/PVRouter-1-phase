32static Channel channelOfAdmux(const PcbVersion board, const uint8_t admux)
33{
34 const uint8_t pin{ static_cast< uint8_t >(admux & MUX_MASK) };
35 for (const Channel ch : chain)
36 {
37 if (AdcChain::pinOf(board, ch) == pin)

39 return ch;

41 }
42 TEST_FAIL_MESSAGE("ADMUX selects an input that is no channel of the board");
43 return Channel::V;

124void test_pipeline_order(void)
125{
126 constexpr uint8_t ROUNDS{ 30 };
127
128 for (const PcbVersion board : boards)
129 {
130 // admuxOf[k]: the ADMUX the conversion of round k runs with
131 uint8_t admuxOf[ROUNDS + 2]{};
132
133 // the chain primed with V, then I_GRID
134 admuxOf[0] = AdcChain::admuxOf(board, Channel::V);
135 admuxOf[1] = AdcChain::admuxOf(board, Channel::I_GRID);
136
137 for (uint8_t round = 0; round != ROUNDS; ++round)
138 {
139 // the ISR of round k believes it handles channelConverted(k)...
140 const Channel believed{ AdcChain::channelConverted(round) };
141
142 // ... and the ADC really converted the channel its ADMUX selected
143 TEST_ASSERT_EQUAL_UINT8(static_cast< uint8_t >(believed), static_cast< uint8_t >(channelOfAdmux(board, admuxOf[round])));
144 TEST_ASSERT_EQUAL_HEX8(AdcChain::refBits(board), admuxOf[round] & REF_MASK);
145 TEST_ASSERT_EQUAL_HEX8(ADLAR_BIT, admuxOf[round] & ADLAR_BIT);
146
147 // the conversion of round k+1 has already started: the write of round k applies to round k+2
148 admuxOf[round + 2] = AdcChain::admuxWrittenAfter(board, believed);
149 }
150 }
151}
