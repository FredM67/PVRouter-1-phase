38inline void begin()
39{
40 Serial.flush(); // no transmit interrupt during the measurement
41 savedTimsk0 = TIMSK0;
42 TIMSK0 = 0; // millis() paused
43 TCCR1B = 0;
44 TCCR1A = 0;
45 TCNT1 = 0;
46 overflows = 0;
47 TIFR1 = _BV(TOV1); // no stale overflow
48 TIMSK1 = _BV(TOIE1);
49 TCCR1B = _BV(CS10); // CPU clock, no prescaler
50}
51
53inline uint32_t end()
54{
55 const uint8_t sreg{ SREG };
56 cli();
57 // read while counting: once stopped, simavr reads TCNT1 as 0
58 const uint16_t low{ TCNT1 };
59 uint16_t high{ overflows };
60 if ((TIFR1 & _BV(TOV1)) && low < 0x8000U)
61 {
62 ++high; // overflow between the last interrupt and the read, not served yet
63 }
64 TCCR1B = 0;
65 TIMSK1 = 0;
66 TIFR1 = _BV(TOV1);
67 TIMSK0 = savedTimsk0;
68 SREG = sreg;
69 return (static_cast< uint32_t >(high) << 16) | low;
70}

73inline void start()
74{
75 if (!calibrated)
76 {
77 calibrated = true;
78 begin();
79 overhead = end();
80 }
81 begin();
82}
83
85inline uint32_t stop()
86{
87 return end() - overhead;
88}
